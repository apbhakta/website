# ============================================================
# Velkommen Inn — Security & Performance (.htaccess)
# UPLOAD INSTRUCTIONS: place this file in the same folder as
# index.html and RENAME it to exactly:  .htaccess
# (the leading dot matters — your file manager has a rename option)
# Works on Apache hosting (GoDaddy, Bluehost, HostGator, etc.)
# ============================================================

# --- Force HTTPS (encrypted connection) ---
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# --- Security headers ---
<IfModule mod_headers.c>
  # Enforce HTTPS for a year once visited
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  # Stop the site being embedded in other sites' frames (clickjacking)
  Header always set X-Frame-Options "SAMEORIGIN"
  # Stop browsers guessing file types
  Header always set X-Content-Type-Options "nosniff"
  # Limit referrer info leaked to other sites
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # Disable powerful browser APIs the site doesn't use
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
</IfModule>

# --- No directory listings ---
Options -Indexes

# --- Block access to hidden/system files ---
<FilesMatch "^\.">
  Require all denied
</FilesMatch>

# --- Compression & caching (speed) ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType text/html "access plus 1 hour"
</IfModule>
